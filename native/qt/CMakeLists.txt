cmake_minimum_required(VERSION 3.25)
project(PhotonShell LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE BOOL "Export the app compile database for clangd" FORCE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Quick Qml QuickControls2 QuickLayouts)
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Qt6 6.9 REQUIRED COMPONENTS DBus)
endif()
if (NOT PHOTON_ENGINE_BUILD_DIR)
    message(FATAL_ERROR "PHOTON_ENGINE_BUILD_DIR must point to photon/build/engine-debug or engine-release")
endif()
find_library(PHOTON_EMBEDDER_LIBRARY NAMES lagom-photonembedder PATHS "${PHOTON_ENGINE_BUILD_DIR}/lib64" "${PHOTON_ENGINE_BUILD_DIR}/lib" NO_DEFAULT_PATH)
if (NOT PHOTON_EMBEDDER_LIBRARY)
    # Setup generates IDE metadata before the first (large) Engine build.
    set(PHOTON_EMBEDDER_LIBRARY "${PHOTON_ENGINE_BUILD_DIR}/lib64/${CMAKE_SHARED_LIBRARY_PREFIX}lagom-photonembedder${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
add_library(Photon::Embedder SHARED IMPORTED GLOBAL)
set_target_properties(Photon::Embedder PROPERTIES IMPORTED_LOCATION "${PHOTON_EMBEDDER_LIBRARY}")
qt_policy(SET QTP0001 NEW)
if (Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
set_source_files_properties(../../ui/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
set_source_files_properties(../../ui/browser/Omnibox.qml PROPERTIES QT_RESOURCE_ALIAS browser/Omnibox.qml)
set_source_files_properties(../../ui/PhotonPage.qml PROPERTIES QT_RESOURCE_ALIAS PhotonPage.qml)
set_source_files_properties(../../ui/components/Button.qml PROPERTIES QT_RESOURCE_ALIAS Button.qml)
set_source_files_properties(../../ui/components/Icon.qml PROPERTIES QT_RESOURCE_ALIAS Icon.qml)
set_source_files_properties(../../ui/components/IconButton.qml PROPERTIES QT_RESOURCE_ALIAS IconButton.qml)
set_source_files_properties(../../ui/components/PressableControl.qml PROPERTIES QT_RESOURCE_ALIAS PressableControl.qml)
set_source_files_properties(../../ui/chrome/TitleBar.qml PROPERTIES QT_RESOURCE_ALIAS TitleBar.qml)
set_source_files_properties(../../ui/pages/WebSurface.qml PROPERTIES QT_RESOURCE_ALIAS WebSurface.qml)
set_source_files_properties(../../ui/theme/Theme.qml PROPERTIES QT_RESOURCE_ALIAS theme/Theme.qml QT_QML_SINGLETON_TYPE TRUE)
file(GLOB_RECURSE PHOTON_ICON_FILES CONFIGURE_DEPENDS ../../ui/icons/*.svg)
foreach(icon_file IN LISTS PHOTON_ICON_FILES)
    file(RELATIVE_PATH icon_resource_path "${CMAKE_CURRENT_SOURCE_DIR}/../../ui/icons" "${icon_file}")
    set_source_files_properties("${icon_file}" PROPERTIES QT_RESOURCE_ALIAS "icons/${icon_resource_path}")
endforeach()
qt_add_executable(photon main.cpp BrowserController.cpp BrowserController.h PhotonAppearance.cpp PhotonAppearance.h PhotonWebView.cpp PhotonWebView.h)
qt_add_qml_module(photon URI Photon VERSION 1.0 SOURCES BrowserController.h QML_FILES ../../ui/Main.qml ../../ui/PhotonPage.qml ../../ui/browser/Omnibox.qml ../../ui/components/Button.qml ../../ui/components/Icon.qml ../../ui/components/IconButton.qml ../../ui/components/PressableControl.qml ../../ui/chrome/TitleBar.qml ../../ui/pages/WebSurface.qml ../../ui/theme/Theme.qml RESOURCES ${PHOTON_ICON_FILES})
target_link_libraries(photon PRIVATE Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::QuickLayouts)
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(photon PRIVATE Qt6::DBus)
endif()
target_include_directories(photon PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Libraries/LibPhotonEmbedder/include")
set(PHOTON_ENGINE_EXPORT_HEADER "${PHOTON_ENGINE_BUILD_DIR}/Libraries/LibPhotonEmbedder/Export.h")
if (NOT EXISTS "${PHOTON_ENGINE_EXPORT_HEADER}")
    # The embedder export header is generated when Engine is configured. Until
    # then, compile the API as an ordinary consumer with an empty export macro.
    set(PHOTON_IDE_INCLUDE_DIR "${CMAKE_BINARY_DIR}/photon_ide/include")
    file(MAKE_DIRECTORY "${PHOTON_IDE_INCLUDE_DIR}/LibPhotonEmbedder")
    file(WRITE "${PHOTON_IDE_INCLUDE_DIR}/LibPhotonEmbedder/Export.h" "#pragma once\n#define PHOTONEMBEDDER_API\n")
    target_include_directories(photon BEFORE PRIVATE "${PHOTON_IDE_INCLUDE_DIR}")
endif()
target_include_directories(photon PRIVATE "${PHOTON_ENGINE_BUILD_DIR}/Libraries")
target_link_libraries(photon PRIVATE Photon::Embedder)
set(PHOTON_RUST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/$<IF:$<CONFIG:Release>,release,debug>/libphoton_core.a")
target_link_libraries(photon PRIVATE "${PHOTON_RUST_LIBRARY}")
if (UNIX AND NOT APPLE)
    target_link_libraries(photon PRIVATE dl pthread m rt util)
endif()
target_compile_definitions(photon PRIVATE PHOTON_HELPER_DIRECTORY="${PHOTON_HELPER_DIRECTORY}")
set_target_properties(photon PROPERTIES BUILD_RPATH "${PHOTON_ENGINE_BUILD_DIR}/lib64;${PHOTON_ENGINE_BUILD_DIR}/vcpkg_installed/x64-linux-dynamic/lib;${PHOTON_ENGINE_BUILD_DIR}/vcpkg_installed/x64-linux-dynamic/debug/lib")

# IDE setup can run before Engine is built; this target compiles the app's real
# C++ translation units without linking the Engine or Rust libraries.
add_custom_target(photon_cpp_check DEPENDS $<TARGET_OBJECTS:photon>)
